\section{Knowledge Synthesis}

\begin{frame}
    \frametitle{Core Concepts Recap}
    \begin{itemize}
        \item Prompt anatomy: role, context, constraints, format, tone.
        \item Model behavior: tokens, attention, capabilities, limitations.
        \item Pattern stacks and frameworks for repeatable success.
        \item Iteration loops: diagnose, critique, refine, document.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Common Pitfalls Revisited}
    \begin{itemize}
        \item Overloaded prompts without clear priorities.
        \item Missing guardrails leading to unsafe or off-brand outputs.
        \item Neglecting evaluation metrics and feedback loops.
        \item Failing to adjust tone/register for target audiences.
    \end{itemize}
\end{frame}

\section{Self-Assessment}

\begin{frame}
    \frametitle{Skills Inventory}
    \begin{itemize}
        \item Rate confidence (1--5) on fundamentals, frameworks, tone control, iteration.
        \item Identify two strengths and two growth areas.
        \item Note resources or support needed (tools, exemplars, peer review).
        \item Share with partner; capture differences in perspective.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Prompting Diagnostic Quiz}
    \begin{itemize}
        \item Short answer: define tokenization, pattern stacking, critique prompt.
        \item Scenario analysis: identify missing components in sample prompts.
        \item Quick build: draft prompt for given task in 5 minutes; peer review.
        \item Discuss answers as a class; clarify misconceptions.
    \end{itemize}
\end{frame}

\section{Integrated Challenges}

\begin{frame}
    \frametitle{Challenge 1: Cross-Domain Brief}
    \begin{itemize}
        \item Teams receive multi-part scenario (e.g., healthcare + policy + comms).
        \item Deliverables: longform summary, patient-facing FAQ, executive brief.
        \item Must reuse prior frameworks, tone guides, and iteration loops.
        \item Submit outputs, critique notes, and final revisions.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Challenge 2: Debugging Lab}
    \begin{itemize}
        \item Provided with flawed AI outputs and prompt history.
        \item Diagnose issues, propose prompt edits, run improved version.
        \item Document fix rationale and residual risks.
        \item Share quick postmortem with class.
    \end{itemize}
\end{frame}
